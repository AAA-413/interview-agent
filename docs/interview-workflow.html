智能面试工作流程

起止
出题（LLM）
追问决策（LLM）
评估（LLM）
总评汇总（LLM）
flowchart TD
    Start(["🚀 用户发起面试"]) --> CheckResume{"有简历?"}

    CheckResume -->|"是"| ParallelGen["并行出题 · 2次LLM"]
    CheckResume -->|"否"| DirectionOnly["方向题出题 · 1次LLM"]

    ParallelGen --> ResumeQ["📋 简历题 LLM · 60%"]
    ParallelGen --> DirectionQ["📖 方向题 LLM · 40%"]
    ResumeQ --> Merge["合并题目列表"]
    DirectionQ --> Merge
    DirectionOnly --> Merge

    Merge --> GetQ["获取当前题目"]
    GetQ --> ShowQ["展示题目给用户"]
    ShowQ --> UserAnswer["用户提交答案"]
    UserAnswer --> SaveAnswer["保存答案"]

    SaveAnswer --> FollowUpCheck{"需要追问?
非追问题 且 追问<2次"}

    FollowUpCheck -->|"是"| FollowUpLLM["🔍 追问决策 LLM"]
    FollowUpLLM --> ShouldFollow{"shouldFollowUp?"}
    ShouldFollow -->|"是"| InsertFollowUp["插入追问题到队列"]
    InsertFollowUp --> GetQ
    ShouldFollow -->|"否"| NextQ["推进到下一题"]

    FollowUpCheck -->|"否"| NextQ

    NextQ --> HasMore{"还有题目?"}
    HasMore -->|"是"| GetQ
    HasMore -->|"否"| TriggerEval["触发评估"]

    TriggerEval --> EvalEach["逐题并发评估
N次LLM · 限流3并发"]
    EvalEach --> EvalBranch{"题型分流"}

    EvalBranch -->|"知识题"| KnowledgeEval["📚 知识题评估
参考答案 + 评分要点"]
    EvalBranch -->|"项目题"| ProjectEval["💼 项目题评估
4维度评分"]

    KnowledgeEval --> SummaryLLM["📊 总评汇总 LLM · 1次"]
    ProjectEval --> SummaryLLM

    SummaryLLM --> Report["生成评估报告"]
    Report --> End(["✅ 展示报告"])
